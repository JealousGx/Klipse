transformers==4.57.6

# LTX-2 (Lightricks/LTX-2) — confirmed real system requirements from the official repo:
# Python >=3.12, CUDA >12.7, PyTorch ~=2.7 (loose declared constraint; see Dockerfile for
# the exact tested version actually pinned). The repo's own setup uses `uv sync` against
# its packages/ workspace (ltx-core, ltx-pipelines) rather than plain PyPI packages —
# the git+subdirectory pip install below is a best-effort translation of that, not
# confirmed working; verify against https://github.com/Lightricks/LTX-2's own setup
# instructions once you have hands-on access, and prefer `uv sync` if pip install fails.
#
# torch + torchaudio + torchvision are installed separately in the Dockerfile from
# PyTorch's own cu128 index, not here from plain PyPI. ltx-core (this repo) depends on
# `torch~=2.7` (pinned) and bare, unpinned `torchaudio`; torchvision isn't declared by
# ltx-core at all but is required at runtime by transformers' AutoImageProcessor. None are
# routed to a CUDA-specific index in ltx-core's own pyproject.toml — so installing them via
# this file's plain-PyPI resolution either drifts to a mismatched CUDA build (confirmed:
# CUDA 13 torchaudio next to a CUDA 12.8 torch, breaking `libcudart.so` loading) or is
# missing entirely (torchvision). PyTorch's versioned cu128 index bundles a fixed, matching
# set for all three.
git+https://github.com/Lightricks/LTX-2.git#subdirectory=packages/ltx-core
git+https://github.com/Lightricks/LTX-2.git#subdirectory=packages/ltx-pipelines
